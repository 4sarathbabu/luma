\subsection{The mean}
\begin{frame}{An equation and its interpretation}
  % Replace example-source with the key of your own entry in references.bib.
  % Optional page locator: \sourcecite[p.~12]{example-source}.
  The mean of $n$ observations is
  \[
    \highlight{\bar{x} = \frac{1}{n}\sum_{i=1}^{n} x_i.}
  \]
  \begin{definitionblock}{Key idea}
    Each observation \highlight{contributes equally} to the average.
  \end{definitionblock}
  \begin{exampleblock}{Example}
    For the observations $2$, $4$, and $6$, the mean is $4$\sourcecite{example-source}.
  \end{exampleblock}
\end{frame}


\begin{frame}{Calculating the mean}
  For the observations $2$, $4$, and $6$:
  % \uncover reserves space so the equation does not move between steps.
  \[
    \begin{aligned}
      \bar{x} &= \frac{1}{n}\sum_{i=1}^{n}x_i \\
      \uncover<2->{&= \frac{2+4+6}{3}} \\
      \uncover<3->{&= \highlight{4}}
    \end{aligned}
  \]
  \uncover<4->{%
    \alert{Interpretation:} The average of these three values is $4$.}
\end{frame}

\begin{frame}{Class discussion}
  \begin{exerciseblock}{Try it yourself}
  How does a single large observation affect the mean?
  \medskip
  \begin{enumerate}
    \item Find the mean of $2$, $4$, and $6$.
    \item Replace $6$ with $24$ and calculate it again.
    \item Explain why the result changes.
  \end{enumerate}
  \end{exerciseblock}
  % Show the question first, then reveal the answer on the next click.
  \begin{exampleblock}<2->{Answer}
    The mean increases from $4$ to $10$ because it uses every value.
  \end{exampleblock}
\end{frame}

\begin{frame}{A common mistake}
  \begin{exerciseblock}{Check the calculation}
    A student computes $(2+4+6)/6$. Is this the mean?
  \end{exerciseblock}
  % The named teaching blocks support the same overlays as Beamer's blocks.
  \begin{cautionblock}<2->{Count the observations}
    Divide by the number of observations, not the largest value.
    Here there are three observations, so the mean is $12/3=4$.
  \end{cautionblock}
\end{frame}
